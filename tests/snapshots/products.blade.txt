// app/Models/Product.php
<?php

namespace App\Models;

use Database\Factories\ProductFactory;
use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\Relations\BelongsToMany;
use Illuminate\Database\Eloquent\Relations\HasMany;
use Illuminate\Database\Eloquent\SoftDeletes;

class Product extends Model
{
    /** @use HasFactory<ProductFactory> */
    use HasFactory;

    use SoftDeletes;

    /**
     * The attributes that are mass assignable.
     *
     * @var list<string>
     */
    protected $fillable = [
        'name',
        'address',
        'slug',
        'description',
        'price',
        'stock',
        'on_sale',
        'features',
        'published_at',
        'expire_at',
        'released_on',
        'sale_time',
        'status',
        'category_id',
        'manufactured_year',
    ];

    /**
     * Get the attributes that should be cast.
     *
     * @return array<string, string>
     */
    protected function casts(): array
    {
        return [
            'price' => 'decimal:2',
            'stock' => 'integer',
            'on_sale' => 'boolean',
            'published_at' => 'datetime',
            'expire_at' => 'datetime',
            'released_on' => 'date',
            'manufactured_year' => 'integer',
        ];
    }

    /**
     * @return BelongsTo<Category, $this>
     */
    public function category(): BelongsTo
    {
        return $this->belongsTo(Category::class);
    }

    /**
     * @return HasMany<AllForeign, $this>
     */
    public function allForeigns(): HasMany
    {
        return $this->hasMany(AllForeign::class, 'product_slug', 'slug');
    }

    /**
     * @return HasMany<Order, $this>
     */
    public function orders(): HasMany
    {
        return $this->hasMany(Order::class, 'product_slug', 'slug');
    }

    /**
     * @return HasMany<SingleValueEnumOrder, $this>
     */
    public function singleValueEnumOrders(): HasMany
    {
        return $this->hasMany(SingleValueEnumOrder::class, 'product_slug', 'slug');
    }

    /**
     * @return BelongsToMany<Tag, $this>
     */
    public function tags(): BelongsToMany
    {
        return $this->belongsToMany(Tag::class);
    }
}

<?php

namespace App\Filters;

use App\Models\Product;
use Javaabu\Filters\DateRangeFilter;
use Javaabu\Filters\Filters;
use Javaabu\Filters\NumberRangeFilter;
use Javaabu\Filters\SelectFilter;
use Javaabu\Filters\Sort;
use Javaabu\Filters\TernaryFilter;
use Javaabu\Filters\TextFilter;
use Javaabu\Filters\TrashedFilter;

class ProductFilters extends Filters
{
    protected ?string $model = Product::class;

    protected ?string $default_sort = 'created_at';

    protected string $default_direction = 'desc';

    protected function filters(): array
    {
        return [
            TextFilter::make('search')->label(__('Search'))->columns(['name', 'address', 'slug', 'description', 'features']),
            NumberRangeFilter::make('price')->label(__('Price')),
            NumberRangeFilter::make('stock')->label(__('Stock'))->integer(),
            TernaryFilter::make('on_sale')->label(__('On sale')),
            SelectFilter::make('status')->label(__('Status'))->options(['draft' => __('Draft'), 'published' => __('Published')]),
            SelectFilter::make('category_id')->label(__('Category'))->relationship('category', 'name')->searchable(),
            SelectFilter::make('tags')->label(__('Tags'))->relationship('tags', 'name')->multiple(),
            DateRangeFilter::make('created_at')->label(__('Created at')),
            TrashedFilter::make(),
        ];
    }

    protected function sorts(): array
    {
        return [
            Sort::make('name')->label(__('Name')),
            Sort::make('address')->label(__('Address')),
            Sort::make('slug')->label(__('Slug')),
            Sort::make('price')->label(__('Price')),
            Sort::make('stock')->label(__('Stock')),
            Sort::make('created_at')->label(__('Created at')),
            Sort::make('updated_at')->label(__('Updated at')),
        ];
    }
}

// app/Http/Controllers/Admin/ProductsController.php
<?php

namespace App\Http\Controllers\Admin;

use App\BulkActions\ProductBulkActions;
use App\Filters\ProductFilters;
use App\Http\Controllers\Controller;
use App\Models\Product;
use App\Stencil\ProductForm;
use Illuminate\Contracts\View\View;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Gate;

class ProductsController extends Controller
{
    /**
     * List the products.
     */
    public function index(Request $request): View
    {
        Gate::authorize('viewAny', Product::class);

        $filters = ProductFilters::make();

        $products = $filters->apply(Product::query())
            ->paginate($request->integer('per_page', 20))
            ->withQueryString();

        return view('admin.products.index', [
            'products' => $products,
            'filters' => $filters,
            'bulk_actions' => ProductBulkActions::make()->paginator($products),
        ]);
    }

    /**
     * Show the form for a new product.
     */
    public function create(): View
    {
        Gate::authorize('create', Product::class);

        return view('admin.products.create', [
            'form' => ProductForm::make()->model(Product::class)->action(route('admin.products.store')),
        ]);
    }

    /**
     * Save a new product.
     */
    public function store(): RedirectResponse
    {
        Gate::authorize('create', Product::class);

        $product = ProductForm::make()->model(Product::class)->save();

        return redirect(route('admin.products.show', $product))
            ->with('status', __('Product created.'));
    }

    /**
     * Show a product.
     */
    public function show(Product $product): View
    {
        Gate::authorize('view', $product);

        $product->load(['category', 'tags']);

        return view('admin.products.show', [
            'product' => $product,
        ]);
    }

    /**
     * Show the form for editing a product.
     */
    public function edit(Product $product): View
    {
        Gate::authorize('update', $product);

        return view('admin.products.edit', [
            'product' => $product,
            'form' => ProductForm::make()->model($product)->action(route('admin.products.update', $product)),
        ]);
    }

    /**
     * Save the changes to a product.
     */
    public function update(Product $product): RedirectResponse
    {
        Gate::authorize('update', $product);

        ProductForm::make()->model($product)->save();

        return redirect(route('admin.products.show', $product))
            ->with('status', __('Product updated.'));
    }

    /**
     * Delete a product.
     */
    public function destroy(Product $product): RedirectResponse
    {
        Gate::authorize('delete', $product);

        $product->delete();

        return redirect(route('admin.products.index'))
            ->with('status', __('Product deleted.'));
    }

    /**
     * Restore a deleted product.
     */
    public function restore(Product $product): RedirectResponse
    {
        Gate::authorize('restore', $product);

        $product->restore();

        return redirect(route('admin.products.show', $product))
            ->with('status', __('Product restored.'));
    }

    /**
     * Permanently delete a product.
     */
    public function forceDelete(Product $product): RedirectResponse
    {
        Gate::authorize('forceDelete', $product);

        $product->forceDelete();

        return redirect(route('admin.products.index'))
            ->with('status', __('Product permanently deleted.'));
    }
}

        $this->assertNotSoftDeleted($product);

        $this->assertModelMissing($product);

    public function it_deletes_products_in_bulk(): void
    {
        $products = Product::factory()->count(2)->create();

        $this->actingAsAdmin('view_products', 'delete_products');

        $this->postJson(route('admin.products.bulk'), [
            'action' => 'delete',
            'products' => $products->pluck('id')->all(),

            ->assertOk()

    /**

    protected function actingAsAdmin(string ...$permissions): User
    {
        $user = User::factory()->create();

        if ($permissions) {
            $user->givePermissionTo($permissions);
        }

        $this->actingAs($user);

        return $user;
    }
