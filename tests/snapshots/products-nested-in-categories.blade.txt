// app/Http/Controllers/Admin/ProductsController.php
<?php

namespace App\Http\Controllers\Admin;

use App\BulkActions\ProductBulkActions;
use App\Filters\ProductFilters;
use App\Http\Controllers\Controller;
use App\Models\Category;
use App\Models\Product;
use App\Stencil\ProductForm;
use Illuminate\Contracts\View\View;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Gate;

class ProductsController extends Controller
{
    /**
     * List the products.
     */
    public function index(Request $request, Category $category): View
    {
        Gate::authorize('viewAny', Product::class);

        $filters = ProductFilters::make();

        $products = $filters->apply(Product::query()->whereBelongsTo($category, 'category'))
            ->paginate($request->integer('per_page', 20))
            ->withQueryString();

        return view('admin.products.index', [
            'category' => $category,
            'products' => $products,
            'filters' => $filters,
            'bulk_actions' => ProductBulkActions::make()->paginator($products),
        ]);
    }

    /**
     * Show the form for a new product.
     */
    public function create(Category $category): View
    {
        Gate::authorize('create', Product::class);

        return view('admin.products.create', [
            'category' => $category,
            'form' => ProductForm::make()->model(Product::class)->action(route('admin.categories.products.store', $category)),
        ]);
    }

    /**
     * Save a new product.
     */
    public function store(Category $category): RedirectResponse
    {
        Gate::authorize('create', Product::class);

        $product = ProductForm::make()->model((new Product)->category()->associate($category))->save();

        return redirect(route('admin.categories.products.show', [$category, $product]))
            ->with('status', __('Product created.'));
    }

    /**
     * Show a product.
     */
    public function show(Category $category, Product $product): View
    {
        Gate::authorize('view', $product);

        $product->load(['tags']);

        return view('admin.products.show', [
            'category' => $category,
            'product' => $product,
        ]);
    }

    /**
     * Show the form for editing a product.
     */
    public function edit(Category $category, Product $product): View
    {
        Gate::authorize('update', $product);

        return view('admin.products.edit', [
            'category' => $category,
            'product' => $product,
            'form' => ProductForm::make()->model($product)->action(route('admin.categories.products.update', [$category, $product])),
        ]);
    }

    /**
     * Save the changes to a product.
     */
    public function update(Category $category, Product $product): RedirectResponse
    {
        Gate::authorize('update', $product);

        ProductForm::make()->model($product)->save();

        return redirect(route('admin.categories.products.show', [$category, $product]))
            ->with('status', __('Product updated.'));
    }

    /**
     * Delete a product.
     */
    public function destroy(Category $category, Product $product): RedirectResponse
    {
        Gate::authorize('delete', $product);

        $product->delete();

        return redirect(route('admin.categories.products.index', $category))
            ->with('status', __('Product deleted.'));
    }

    /**
     * Restore a deleted product.
     */
    public function restore(Category $category, Product $product): RedirectResponse
    {
        Gate::authorize('restore', $product);

        $product->restore();

        return redirect(route('admin.categories.products.show', [$category, $product]))
            ->with('status', __('Product restored.'));
    }

    /**
     * Permanently delete a product.
     */
    public function forceDelete(Category $category, Product $product): RedirectResponse
    {
        Gate::authorize('forceDelete', $product);

        $product->forceDelete();

        return redirect(route('admin.categories.products.index', $category))
            ->with('status', __('Product permanently deleted.'));
    }
}

// app/BulkActions/ProductBulkActions.php
<?php

namespace App\BulkActions;

use App\Filters\ProductFilters;
use App\Models\Category;
use App\Models\Product;
use Javaabu\BulkActions\Actions\DeleteAction;
use Javaabu\BulkActions\Actions\ForceDeleteAction;
use Javaabu\BulkActions\Actions\RestoreAction;
use Javaabu\BulkActions\BulkActions;

class ProductBulkActions extends BulkActions
{
    protected ?string $model = Product::class;

    protected function setUp(): void
    {
        $this->filters(ProductFilters::class);

        // only the products of the category in the URL
        $this->query(fn () => Product::query()->whereBelongsTo($this->category(), 'category'));
        $this->url(fn () => route('admin.categories.products.bulk', $this->category()));
    }

    protected function actions(): array
    {
        return [
            DeleteAction::make()->can('delete_products'),
            RestoreAction::make()->can('delete_products'),
            ForceDeleteAction::make()->can('force_delete_products'),
        ];
    }

    /**
     * The category in the URL.
     */
    protected function category(): Category
    {
        $category = request()->route('category');

        if ($category instanceof Category) {
            return $category;
        }

        return Category::query()->where('id', $category)->firstOrFail();
    }
}

<?php

namespace Tests\Feature\Admin;

use App\Models\Category;
use App\Models\Product;
use App\Models\Tag;
use App\Models\User;
use Illuminate\Foundation\Testing\RefreshDatabase;
use PHPUnit\Framework\Attributes\Test;
use Tests\TestCase;

class ProductsControllerTest extends TestCase
{
    use RefreshDatabase;

    protected function setUp(): void
    {
        parent::setUp();

        $this->artisan('permissions:sync');
    }

    #[Test]
    public function it_lists_the_products(): void
    {
        $category = Category::factory()->create();
        $product = Product::factory()->for($category, 'category')->create();

        $this->actingAsAdmin('view_products');

        $this->get(route('admin.categories.products.index', $category))
            ->assertOk()
            ->assertSee($product->name);
    }

    #[Test]
    public function it_does_not_list_the_products_without_permission(): void
    {
        $category = Category::factory()->create();
        $this->actingAsAdmin();

        $this->get(route('admin.categories.products.index', $category))
            ->assertForbidden();
    }

    #[Test]
    public function it_searches_the_products(): void
    {
        $category = Category::factory()->create();
        $product = Product::factory()->for($category, 'category')->create(['name' => 'ރަސްމީ ދުވަސް']);
        $other = Product::factory()->for($category, 'category')->create(['name' => 'Something else']);

        $this->actingAsAdmin('view_products');

        $this->get(route('admin.categories.products.index', [$category, 'search' => 'ރަސްމީ']))
            ->assertOk()
            ->assertSee('ރަސްމީ ދުވަސް')
            ->assertDontSee('Something else');
    }

    #[Test]
    public function it_shows_the_new_product_form(): void
    {
        $category = Category::factory()->create();
        $this->actingAsAdmin('edit_products');

        $this->get(route('admin.categories.products.create', $category))
            ->assertOk()
            ->assertSee(__('New product'));
    }

    #[Test]
    public function it_creates_a_product(): void
    {
        $category = Category::factory()->create();
        $tags = Tag::factory()->count(2)->create();

        $this->actingAsAdmin('edit_products');

        $this->post(route('admin.categories.products.store', $category), [
            'name' => 'ރަސްމީ',
            'price' => 10.5,
            'status' => 'draft',
            'tags' => $tags->pluck('id')->all(),
        ])
            ->assertRedirect()
            ->assertSessionHasNoErrors();

        $this->assertDatabaseHas('products', [
            'category_id' => $category->id,
            'name' => 'ރަސްމީ',
            'price' => 10.5,
            'status' => 'draft',
        ]);

        $product = Product::query()->latest('id')->first();

        $this->assertEquals($tags->pluck('id')->all(), $product->tags()->pluck('tags.id')->sort()->values()->all());
    }

    #[Test]
    public function it_validates_new_products(): void
    {
        $category = Category::factory()->create();
        $this->actingAsAdmin('edit_products');

        $this->post(route('admin.categories.products.store', $category), [
            'name' => [],
            'price' => 'ބަދަލު',
            'status' => 'ބަދަލު',
        ])
            ->assertSessionHasErrors(['name', 'price', 'status']);

        $this->assertDatabaseCount('products', 0);
    }

    #[Test]
    public function it_does_not_create_products_without_permission(): void
    {
        $category = Category::factory()->create();
        $this->actingAsAdmin('view_products');

        $this->post(route('admin.categories.products.store', $category), [])
            ->assertForbidden();

        $this->assertDatabaseCount('products', 0);
    }

    #[Test]
    public function it_shows_a_product(): void
    {
        $category = Category::factory()->create();
        $product = Product::factory()->for($category, 'category')->create();

        $this->actingAsAdmin('view_products');

        $this->get(route('admin.categories.products.show', [$category, $product]))
            ->assertOk()
            ->assertSee($product->name);
    }

    #[Test]
    public function it_shows_the_edit_product_form(): void
    {
        $category = Category::factory()->create();
        $product = Product::factory()->for($category, 'category')->create();

        $this->actingAsAdmin('edit_products');

        $this->get(route('admin.categories.products.edit', [$category, $product]))
            ->assertOk()
            ->assertSee(__('Edit product'));
    }

    #[Test]
    public function it_updates_a_product(): void
    {
        $category = Category::factory()->create();
        $tags = Tag::factory()->count(2)->create();

        $product = Product::factory()->for($category, 'category')->create();

        $this->actingAsAdmin('edit_products');

        $this->patch(route('admin.categories.products.update', [$category, $product]), [
            'name' => 'ދިވެހި',
            'price' => 20.25,
            'status' => 'published',
            'tags' => [$tags->first()->id],
        ])
            ->assertRedirect()
            ->assertSessionHasNoErrors();

        $this->assertDatabaseHas('products', [
            'id' => $product->id,
            'name' => 'ދިވެހި',
            'price' => 20.25,
            'status' => 'published',
        ]);

        $product->refresh();

        $this->assertEquals([$tags->first()->id], $product->tags()->pluck('tags.id')->sort()->values()->all());
    }

    #[Test]
    public function it_does_not_update_products_without_permission(): void
    {
        $category = Category::factory()->create();
        $product = Product::factory()->for($category, 'category')->create();

        $this->actingAsAdmin('view_products');

        $this->patch(route('admin.categories.products.update', [$category, $product]), [])
            ->assertForbidden();
    }

    #[Test]
    public function it_deletes_a_product(): void
    {
        $category = Category::factory()->create();
        $product = Product::factory()->for($category, 'category')->create();

        $this->actingAsAdmin('delete_products');

        $this->delete(route('admin.categories.products.destroy', [$category, $product]))
            ->assertRedirect(route('admin.categories.products.index', $category));

        $this->assertSoftDeleted($product);
    }

    #[Test]
    public function it_restores_a_deleted_product(): void
    {
        $category = Category::factory()->create();
        $product = Product::factory()->for($category, 'category')->create();
        $product->delete();

        $this->actingAsAdmin('delete_products');

        $this->patch(route('admin.categories.products.restore', [$category, $product]))
            ->assertRedirect();

        $this->assertNotSoftDeleted($product);
    }

    #[Test]
    public function it_permanently_deletes_a_product(): void
    {
        $category = Category::factory()->create();
        $product = Product::factory()->for($category, 'category')->create();
        $product->delete();

        $this->actingAsAdmin('force_delete_products');

        $this->delete(route('admin.categories.products.force-delete', [$category, $product]))
            ->assertRedirect(route('admin.categories.products.index', $category));

        $this->assertModelMissing($product);
    }

    #[Test]
    public function it_deletes_products_in_bulk(): void
    {
        $category = Category::factory()->create();
        $products = Product::factory()->for($category, 'category')->count(2)->create();

        $this->actingAsAdmin('view_products', 'delete_products');

        $this->postJson(route('admin.categories.products.bulk', $category), [
            'action' => 'delete',
            'products' => $products->pluck('id')->all(),
        ])
            ->assertOk()
            ->assertJson(['succeeded' => 2]);

        $products->each(fn (Product $record) => $this->assertSoftDeleted($record));
    }

    /**
     * Log in as a user of the admin area with these permissions.
     */
    protected function actingAsAdmin(string ...$permissions): User
    {
        $user = User::factory()->create();

        if ($permissions) {
            $user->givePermissionTo($permissions);
        }

        $this->actingAs($user);

        return $user;
    }
}
