// app/Models/Post.php
<?php

namespace App\Models;

use Database\Factories\PostFactory;
use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Javaabu\Mediapicker\Concerns\InteractsWithAttachments;
use Javaabu\Mediapicker\Contracts\HasAttachments;
use Javaabu\Mediapicker\Support\FileTypes;
use Spatie\MediaLibrary\MediaCollections\File;

class Post extends Model implements HasAttachments
{
    /** @use HasFactory<PostFactory> */
    use HasFactory;

    use InteractsWithAttachments;

    /**
     * The attributes that are mass assignable.
     *
     * @var list<string>
     */
    protected $fillable = [
        'title',
        'published_at',
        'excerpt',
        'content',
        'medium_content',
        'tiny_content',
    ];

    /**
     * Get the attributes that should be cast.
     *
     * @return array<string, string>
     */
    protected function casts(): array
    {
        return [
            'published_at' => 'datetime',
        ];
    }

    /**
     * The media collections, picked from the media library.
     */
    public function registerAttachmentCollections(): void
    {
        $this->addAttachmentCollection('featured_image')
                ->singleFile()
            ->acceptsFile(fn (File $file) => FileTypes::isAllowed($file->mimeType, 'image'));

        $this->addAttachmentCollection('gallery')
            ->acceptsFile(fn (File $file) => FileTypes::isAllowed($file->mimeType, 'image'));

        $this->addAttachmentCollection('attachments')
            ->acceptsFile(fn (File $file) => FileTypes::isAllowed($file->mimeType, 'document'));
    }
}

// app/Stencil/PostForm.php
<?php

namespace App\Stencil;

use Javaabu\Mediapicker\Stencil\MediaPickerField;
use Javaabu\Stencil\Components\Section;
use Javaabu\Stencil\Fields\DateTimePicker;
use Javaabu\Stencil\Fields\Textarea;
use Javaabu\Stencil\Fields\TextInput;
use Javaabu\Stencil\Form;

class PostForm extends Form
{
    protected function definition(): array
    {
        return [
            Section::make(__('Post'))
                ->columns(2)
                ->schema([
                    TextInput::make('title')
                        ->label(__('Title'))
                        ->required()
                        ->maxLength(255),
                    DateTimePicker::make('published_at')->label(__('Published at')),
                    Textarea::make('excerpt')
                        ->label(__('Excerpt'))
                        ->required()
                        ->rows(4)
                        ->maxLength(65535)
                        ->columnSpanFull(),
                    Textarea::make('content')
                        ->label(__('Content'))
                        ->rows(4)
                        ->maxLength(65535)
                        ->columnSpanFull(),
                    Textarea::make('medium_content')
                        ->label(__('Medium content'))
                        ->rows(4)
                        ->maxLength(65535)
                        ->columnSpanFull(),
                    Textarea::make('tiny_content')
                        ->label(__('Tiny content'))
                        ->rows(4)
                        ->maxLength(65535)
                        ->columnSpanFull(),
                ]),
            Section::make(__('Media'))
                ->schema([
                    MediaPickerField::make('featured_image')
                        ->label(__('Featured image'))
                        ->mediaType('image'),
                    MediaPickerField::make('gallery')
                        ->label(__('Gallery'))
                        ->mediaType('image')
                        ->multiple(),
                    MediaPickerField::make('attachments')
                        ->label(__('Attachments'))
                        ->mediaType('document')
                        ->multiple(),
                ]),
        ];
    }
}
